cmake_minimum_required(VERSION 3.16)
project(sim_relay VERSION 0.6.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
add_compile_options(-Wall -Wextra -Wpedantic -Werror -ffunction-sections -fdata-sections)
set(RELAY_CORE src/modemmanager.c src/lite.c src/adapter.c src/protocol.c src/zte_send.c src/vault.c src/zte_operations.c src/push_config.c src/api.c)
add_executable(sim-relayd src/main.c src/http.c src/push_transport.c ${RELAY_CORE})
target_compile_definitions(sim-relayd PRIVATE _POSIX_C_SOURCE=200809L)
target_link_libraries(sim-relayd PRIVATE curl cjson sodium)
target_link_options(sim-relayd PRIVATE -Wl,--gc-sections)
install(TARGETS sim-relayd RUNTIME DESTINATION sbin)
option(SIM_RELAY_TESTS "Build offline protocol tests" OFF)
if(SIM_RELAY_TESTS)
  enable_testing()
  add_executable(test-decode tests/test_decode.c src/http.c src/modemmanager.c src/lite.c src/protocol.c src/zte_send.c src/vault.c src/zte_operations.c src/push_config.c src/push_transport.c)
  target_compile_options(test-decode PRIVATE -UNDEBUG)
  target_compile_definitions(test-decode PRIVATE _POSIX_C_SOURCE=200809L)
  target_link_libraries(test-decode PRIVATE curl cjson sodium)
  add_test(NAME adapter-decode COMMAND test-decode)
  add_executable(test-push-format tests/test_push_format.c src/http.c ${RELAY_CORE})
  target_compile_options(test-push-format PRIVATE -UNDEBUG)
  target_compile_definitions(test-push-format PRIVATE _POSIX_C_SOURCE=200809L)
  target_link_libraries(test-push-format PRIVATE curl cjson sodium)
  add_test(NAME push-format COMMAND test-push-format)
  add_executable(test-http-headers tests/test_http_headers.c)
  target_compile_options(test-http-headers PRIVATE -UNDEBUG)
  target_compile_definitions(test-http-headers PRIVATE _POSIX_C_SOURCE=200809L)
  target_link_libraries(test-http-headers PRIVATE curl)
  add_test(NAME http-header-allocation COMMAND test-http-headers)
  add_executable(sim-relayd-fixture src/main.c tests/fixture_http.c tests/fixture_push.c ${RELAY_CORE})
  target_compile_definitions(sim-relayd-fixture PRIVATE _POSIX_C_SOURCE=200809L RELAY_TESTING=1)
  target_link_libraries(sim-relayd-fixture PRIVATE curl cjson sodium)
  find_program(NODE_EXECUTABLE node)
  if(NODE_EXECUTABLE)
    add_test(NAME luci-rpc-flows COMMAND ${NODE_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_luci.js)
    add_test(NAME luci-rpc-flows-zh-cn COMMAND ${NODE_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_luci.js --zh-cn)
  endif()
endif()
